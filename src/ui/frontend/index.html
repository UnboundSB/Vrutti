#include <iostream>
#include <thread>

namespace vrutti::core::ipc {

    void IPCClient::start() {
        if (m_running) return;
        m_running = true;
        
        std::thread([this]() { this->listenLoop(); }).detach();
        std::cout << "[IPC] Bound to pipe/socket: " << m_pipeName << std::endl;
    }

} // namespace vrutti::core::ipc
